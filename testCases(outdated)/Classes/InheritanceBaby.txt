//===================TEST ONE: basic call no 'this'=====================//
#pullf io
class Math
{
    make int add(int a, int b)
    {
        return a + b;
    }
    make int multiply(int a, int b)
    {
        return a * b;
    }
}
make Math m = Math();
make int sum = m.add(5, 3);
make int product = m.multiply(4, 6);
println(sum);
println(product);
// Expected: 8, 24

//===================TEST NINE: multiple instances of same subclass, field independence=====================//
#pullf io
class Counter
{
    make int count = 0;
    make empty setCount(int n)
    {
        this.count = n;
    }
}
class NamedCounter -> Counter
{
    make string label = "default";
    make empty setLabel(string l)
    {
        this.label = l;
    }
}
make NamedCounter c1 = NamedCounter();
make NamedCounter c2 = NamedCounter();
c1.setCount(5);
c1.setLabel("first");
c2.setCount(99);
c2.setLabel("second");
println(c1.count);
println(c1.label);
println(c2.count);
println(c2.label);
// Expected: 5, first, 99, second


//===================TEST TEN: constructor field inheritance + inherited method reading subclass-set field=====================//
#pullf io
class Person
{
    make string name = "";
    make empty Person(string n)
    {
        this.name = n;
    }
    make empty introduce()
    {
        println("Hi, I'm");
        println(this.name);
    }
}
class Student -> Person
{
    make string school = "";
    make empty Student(string n, string s)
    {
        this.name = n;
        this.school = s;
    }
}
make Student s = Student("Alice", "MIT");
s.introduce();
println(s.school);
// Expected: Hi, I'm, Alice, MIT

//===================TEST THIRTEEN: this-dispatch through inherited AND subclass-defined methods=====================//
#pullf io
class Calculator
{
    make int square(int n)
    {
        return this.multiplyBySelf(n);
    }
    make int multiplyBySelf(int n)
    {
        return n * n;
    }
}
class ScientificCalculator -> Calculator
{
    make int cube(int n)
    {
        return n * this.square(n);
    }
}
make ScientificCalculator sc = ScientificCalculator();
println(sc.square(4));
println(sc.cube(3));
// Expected: 16, 27
